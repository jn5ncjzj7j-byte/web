function doGet(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("ชีต1"); var data = sheet.getDataRange().getValues(); // กรณีค้นหาชื่อ (Search) if (e.parameter.name) { for (var i = 1; i < data.length; i++) { if (data[i][0] == e.parameter.name) { var result = { 'amount': data[i][1] || 0, 'balance': data[i][2] || 0 }; return ContentService.createTextOutput(JSON.stringify(result)) .setMimeType(ContentService.MimeType.JSON); } } } // กรณีโหลดชื่อทั้งหมด (Dropdown) var names = []; for (var i = 1; i < data.length; i++) { if(data[i][0]) names.push(data[i][0]); } return ContentService.createTextOutput(JSON.stringify(names)) .setMimeType(ContentService.MimeType.JSON); }
